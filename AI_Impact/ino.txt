void loop() {
  // Handle connection monitoring
  if (connectionInProgress) {
    monitorConnection();
  }

  // Update sensor values
  if (millis() - lastSensorRead > 100) {
    readAllSensors();
    printIndividualSensorData();
    lastSensorRead = millis();
    if ((millis() - lastWsBroadcast) >= wsBroadcastIntervalMs && ws.count() > 0) {
      ws.textAll(buildSensorJson());
      lastWsBroadcast = millis();
    }
  }

  // Status updates
  if (millis() - lastStatusCheck > 5000) {
    printStatus();
    lastStatusCheck = millis();
  }

  // Check WiFi health in sensor mode
  if (isConnectedToHotspot && !connectionInProgress) {
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println("WiFi disconnected - returning to provisioning mode");
      returnToProvisioningMode();
    }
  }

  delay(100);
  yield();
}

void readAllSensors() {
  // Battery voltage
  sensors.batteryVoltage = readBatteryVoltage();

  // Basic sensors
  int ldrValue = analogRead(LDR_PIN);
  sensors.lux = ldrValue * (600.0 / 4095.0);

  sensors.gasRatio = (digitalRead(gasSensorPin) == LOW) ? 100 : 0;

  int raindropValue = analogRead(RaindropPin);
  sensors.Raindrop = map(raindropValue, 0, 4095, 0, 100);
  // DHT11
  sensors.humidity = dht.readHumidity();
  sensors.temperature = dht.readTemperature();
  if (isnan(sensors.humidity)) sensors.humidity = 0;
  if (isnan(sensors.temperature)) sensors.temperature = 0;
  // DS18B20
  ds18b20.requestTemperatures();
  sensors.liquidTemp = ds18b20.getTempCByIndex(0);
  if (sensors.liquidTemp == DEVICE_DISCONNECTED_C) sensors.liquidTemp = 0;
  // Vibration
  int vibrationValue = analogRead(vibrationPin);
  sensors.vibrationLevel = map(vibrationValue, 0, 1023, 0, 20);
  Serial.println(vibrationValue);
  // Ultrasonic
  // digitalWrite(trigPin, LOW);
  // delayMicroseconds(2);

  // delayMicroseconds(10);
  // digitalWrite(trigPin, LOW);
  // long duration = pulseIn(echoPin, HIGH, 30000);
  // sensors.distance = duration * 0.034 / 2;
  delay(50);
  sensors.distance = sonar.ping_cm();
  int int_distance = int(sensors.distance);
  // if (sensors.distance > 200 || sensors.distance == 0) {
  //   server.send(200, "text/plain", "max distance is reached");

  //   server.send(200, "text/plain", String(int_distance));

  int soundValue = analogRead(soundPin);
  sensors.soundLevel = map(soundValue, 0, 1023, 0, 130);

  if (digitalRead(pirPin) == HIGH) {
    if (startTime == 0) {
      startTime = millis();
    }

  readMAX30105();
}

void readSpeedSensor() {
  sensor1State = digitalRead(sensorPin1);
  sensor2State = digitalRead(sensorPin2);

  if (sensor1State == 0 && speedStartTime == 0) {
    speedStartTime = millis();
    while (digitalRead(sensorPin2)) {
      if (millis() - speedStartTime > speedTimeout) {
        sensors.speedKmph = 0;
        speedStartTime = 0;
        return;
      }
      delay(3);
    }

    speedInterval = millis() - speedStartTime;
    if (speedInterval > 3) {
      sensors.speedKmph = ((dist) / (speedInterval / 1000.0)) * (18 / 5.0);
    }
    speedStartTime = 0;
  } else if (millis() - speedStartTime > speedTimeout && speedStartTime != 0) {
    speedStartTime = 0;
    sensors.speedKmph = 0;
  }
}

void readMAX30105() {
  if (!max30105Initialized) {
    //SerialBT.println("BPM(Beat per minute): 0");
    Serial.println("BPM(Beat per minute): 0");
    sensors.heartRate = 0;
    sensors.spO2 = 0;
    sensors.bodyTempF = 32;
    return;
  }

  // Heart rate reading
  long irValue = particleSensor.getIR();
  Serial.println(irValue);
  if (checkForBeat(irValue)) {
    long delta = millis() - lastBeat;
    lastBeat = millis();
    Serial.println("Beat Detected!");
    beatsPerMinute = 60 / (delta / 1000.0);

    if (beatsPerMinute < 255 && beatsPerMinute > 20) {
      rates[rateSpot++] = (byte)beatsPerMinute;
      rateSpot %= RATE_SIZE;

      beatAvg = 0;
      for (byte x = 0; x < RATE_SIZE; x++) {
        beatAvg += rates[x];
      }
      beatAvg /= RATE_SIZE;
      Serial.print("Average BPM = ");
      Serial.println(beatAvg);
    }
  }

  if (irValue < 50000) {
    sensors.heartRate = 0;
  } else {
    sensors.heartRate = beatAvg;
  }
  // SpO2 calculation
  uint32_t red = particleSensor.getFIFORed();
  uint32_t ir = particleSensor.getFIFOIR();

  if (red > 0 && ir > 0) {
    double fred = (double)red;
    double fir = (double)ir;
    avered = avered * frate + fred * (1.0 - frate);
    aveir = aveir * frate + fir * (1.0 - frate);
    sumredrms += (fred - avered) * (fred - avered);
    sumirrms += (fir - aveir) * (fir - aveir);

    spo2_i++;
    if ((spo2_i % 100) == 0) {
      double R = (sqrt(sumredrms) / avered) / (sqrt(sumirrms) / aveir);
      double SpO2 = -23.3 * (R - 0.4) + 100;
      ESpO2 = FSpO2 * ESpO2 + (1.0 - FSpO2) * SpO2;
      sensors.spO2 = ESpO2;
      sumredrms = 0.0;
      sumirrms = 0.0;
      spo2_i = 0;
    }
  }
  // Temperature reading
  sensors.bodyTempF = particleSensor.readTemperatureF();
  if (isnan(sensors.bodyTempF)) sensors.bodyTempF = 32;
}

float readBatteryVoltage() {
  int rawValue = analogRead(batteryPin);
  float voltage = (float)rawValue / 4095.0 * 3.7;
  actualVoltage = voltage * (R1 + R2) / R2;
  return actualVoltage;
}

// ---- Logging helpers ----
void logLine(const String &line) {
  logBuffer[(logHead + logSize) % LOG_CAPACITY] = line;
  if (logSize < LOG_CAPACITY) {
    logSize++;
  } else {
    logHead = (logHead + 1) % LOG_CAPACITY;
  }
}

String getLogsJson() {
  String json = "{\"lines\":[";
  for (int i = 0; i < logSize; i++) {
    if (i) json += ",";
    String line = logBuffer[(logHead + i) % LOG_CAPACITY];
    line.replace("\\", "\\\\");
    line.replace("\"", "\\\"");
    line.replace("\n", "");
    json += "\"" + line + "\"";
  }
  json += "]}";
  return json;
}

void startProvisioningMode() {
  Serial.println("Starting Dual Mode (AP+STA)");
  logLine("Dual mode starting");
  setBlue();  // Blue for provisioning mode

  isProvisioningMode = true;
  connectionInProgress = false;
  connectionRetries = 0;

  WiFi.disconnect(true);
  WiFi.mode(WIFI_OFF);
  delay(1000);

  // Configure ESP32 in AP+STA mode for dual network operation
  WiFi.mode(WIFI_AP_STA);
  delay(1000);

  // Configure Access Point with fixed IP
  IPAddress local_ip(192, 168, 4, 1);  // AP IP address
  IPAddress gateway(192, 168, 4, 1);   // AP gateway
  IPAddress subnet(255, 255, 255, 0);  // Subnet mask

  WiFi.softAPConfig(local_ip, gateway, subnet);

  // Start SoftAP with SSID and password
  if (WiFi.softAP("AI spark-100", "12345678", 1, 0, 4)) {
    Serial.println(" Access Point started successfully");
    logLine("AP started");
  } else {
    Serial.println(" Failed to start Access Point");
    logLine("AP failed");
    setRed();
    delay(5000);
    return;
  }

  delay(1000);
  apIP = WiFi.softAPIP();

  Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");

void monitorConnection() {
  if (!WiFi.isConnected() && (millis() - connectionStartTime > 5000)) {
    if (connectionRetries < maxRetries) {
      attemptWiFiConnection();
    } else {
      Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
      Serial.println(" All STA connection attempts failed");
      Serial.println("AP still active at: " + apIP.toString());
      Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
      logLine("STA connection failed");
      connectionInProgress = false;
      connectionRetries = 0;
      setBlue();  // Back to blue (AP only mode)
      return;
    }
  }

  if (WiFi.status() == WL_CONNECTED && connectionInProgress) {
    staIP = WiFi.localIP();
    isConnectedToHotspot = true;
    isProvisioningMode = false;
    connectionInProgress = false;

    Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
    Serial.println(" DUAL MODE ACTIVE (AP+STA)");
    Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
    Serial.println("AP IP:  " + apIP.toString());
    Serial.println("STA IP: " + staIP.toString());
    Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
    Serial.println("Access sensor dashboard:");
    Serial.println("  • Via AP:  http://" + apIP.toString());
    Serial.println("  • Via STA: http://" + staIP.toString());
    Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
    logLine("Dual mode: STA IP " + staIP.toString());
    setGreen();  // Green for connected

    // Setup sensor endpoints (web server already running for provision page)
    delay(1000);
    setupSensorServer();
  }

  if (connectionInProgress && (millis() - connectionStartTime > 30000)) {
    Serial.println(" Connection attempt timed out");
    logLine("Connection timeout");
    WiFi.disconnect();
    connectionRetries++;
    connectionStartTime = millis();

    if (connectionRetries >= maxRetries) {
      connectionInProgress = false;
      setBlue();  // Back to AP mode
    }
  }
}
void attemptWiFiConnection() {
  Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
  Serial.println("Connection Attempt " + String(connectionRetries + 1) + "/" + String(maxRetries));
  Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");

  // In dual mode, keep AP active - only initiate STA connection
  // No need to close the AP or change WiFi mode (already in WIFI_AP_STA)

  Serial.println("Connecting to: " + enteredSSID);
  Serial.println("AP remains active at: " + apIP.toString());
  logLine("Connecting to " + enteredSSID);

  WiFi.begin(enteredSSID.c_str(), enteredPASS.c_str());

  connectionRetries++;
  connectionStartTime = millis();
}
